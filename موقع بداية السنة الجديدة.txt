<script>
"use strict";

/* ====== عدّل هذين السطرين فقط ====== */
const MY_NAME = "";                    // اكتب اسمك هنا ليظهر أسفل الصفحة
const DEFAULT_TEXT = "عام دراسي جديد"; // الكلمة التي تظهر في البداية
/* =================================== */

const TIPS = [
  "ذاكر على دفعات قصيرة: خمس وعشرون دقيقة تركيز، ثم خمس دقائق راحة.",
  "اشرح الدرس بصوت عالٍ كأنك تعلّمه لصديق؛ ما لا تستطيع شرحه لم تفهمه بعد.",
  "نم مبكراً قبل الاختبار، فالدماغ يثبّت المعلومات أثناء النوم.",
  "اكتب أسئلتك بجانب كل درس، وأجب عنها بعد يومين دون النظر إلى الكتاب.",
  "ابدأ بالمادة الأصعب حين يكون تركيزك في أعلى مستوى.",
  "رتّب حقيبتك في المساء، فالصباح الهادئ يبدأ من ليلة منظّمة.",
  "اسأل حين لا تفهم؛ السؤال الجيد يفيد الصف كله.",
  "قسّم المشروع الكبير إلى خطوات صغيرة، وضع لكل خطوة موعداً.",
  "راجع ما تعلّمته اليوم لخمس دقائق قبل النوم.",
  "أبعد هاتفك عن مكتبك أثناء المذاكرة؛ وجوده وحده يشتّت الانتباه.",
  "خصّص لكل مادة لوناً أو دفتراً، فالتنظيم يوفّر عليك وقتاً كبيراً.",
  "الخطأ جزء من التعلّم؛ دوّن أخطاءك وراجعها قبل الاختبار."
];

const $ = (id) => document.getElementById(id);
const canvas = $("stage");
const ctx = canvas.getContext("2d");
const off = document.createElement("canvas");
const octx = off.getContext("2d", { willReadFrequently: true });
const reduceMotion = matchMedia("(prefers-reduced-motion: reduce)").matches;
const FAMILY = 'Lalezar, Tajawal, "Segoe UI", Tahoma, sans-serif';

const INK = "#14213D";
const HIGHLIGHT = "#F2B705";

let W = 0, H = 0, dpr = 1, dotR = 2;
let parts = [];
let text = DEFAULT_TEXT;
let running = false;
const mouse = { x: -9999, y: -9999, power: 0, base: 0 };

function fitCanvas() {
  const r = canvas.getBoundingClientRect();
  dpr = Math.min(window.devicePixelRatio || 1, 2);
  W = Math.max(1, Math.round(r.width * dpr));
  H = Math.max(1, Math.round(r.height * dpr));
  canvas.width = off.width = W;
  canvas.height = off.height = H;
  mouse.base = 7 * dpr;
  mouse.power = mouse.base;
}

/* تحويل النص إلى نقاط: نرسمه على لوحة مخفية ثم نأخذ عيّنات من البكسلات */
function sampleTargets() {
  octx.clearRect(0, 0, W, H);
  octx.fillStyle = "#000";
  octx.textAlign = "center";
  octx.textBaseline = "middle";
  octx.direction = "rtl";

  let size = H * 0.66;
  octx.font = size + "px " + FAMILY;
  const maxW = W * 0.9;
  const w = octx.measureText(text).width;
  if (w > maxW) {
    size *= maxW / w;
    octx.font = size + "px " + FAMILY;
  }
  octx.fillText(text, W / 2, H / 2 + size * 0.04);

  const data = octx.getImageData(0, 0, W, H).data;
  let gap = Math.max(3 * dpr, size / 30);
  let pts;
  for (;;) {                       // نزيد المسافة إن كثرت النقاط حفاظاً على السرعة
    pts = [];
    for (let y = 0; y < H; y += gap) {
      for (let x = 0; x < W; x += gap) {
        if (data[(Math.floor(y) * W + Math.floor(x)) * 4 + 3] > 128) pts.push(x, y);
      }
    }
    if (pts.length / 2 <= 3200) break;
    gap *= 1.15;
  }
  dotR = Math.max(1.2 * dpr, gap * 0.3);

  const n = pts.length / 2;
  while (parts.length < n) {
    parts.push({
      x: Math.random() * W, y: Math.random() * H,
      vx: 0, vy: 0, tx: 0, ty: 0, hi: Math.random() < 0.12
    });
  }
  parts.length = n;
  for (let i = 0; i < n; i++) { parts[i].tx = pts[i * 2]; parts[i].ty = pts[i * 2 + 1]; }

  canvas.setAttribute("aria-label", text);
  if (reduceMotion) {
    parts.forEach((p) => { p.x = p.tx; p.y = p.ty; });
    draw();
  }
}

function draw() {
  ctx.clearRect(0, 0, W, H);
  for (const pass of [false, true]) {
    ctx.fillStyle = pass ? HIGHLIGHT : INK;
    ctx.beginPath();
    for (const p of parts) {
      if (p.hi !== pass) continue;
      ctx.moveTo(p.x + dotR, p.y);
      ctx.arc(p.x, p.y, dotR, 0, 6.2832);
    }
    ctx.fill();
  }
}

function tick() {
  const R = 95 * dpr, R2 = R * R;
  for (const p of parts) {
    let ax = (p.tx - p.x) * 0.055;
    let ay = (p.ty - p.y) * 0.055;
    const dx = p.x - mouse.x, dy = p.y - mouse.y, d2 = dx * dx + dy * dy;
    if (d2 < R2 && d2 > 0.01) {
      const d = Math.sqrt(d2), f = (1 - d / R) * mouse.power;
      ax += (dx / d) * f;
      ay += (dy / d) * f;
    }
    p.vx = (p.vx + ax) * 0.84;
    p.vy = (p.vy + ay) * 0.84;
    p.x += p.vx;
    p.y += p.vy;
  }
  mouse.power += (mouse.base - mouse.power) * 0.06;   // تهدأ قوة الانفجار تدريجياً
  draw();
  requestAnimationFrame(tick);
}

async function loadFont() {
  try {
    if (document.fonts && document.fonts.load) {
      await Promise.race([
        document.fonts.load("100px Lalezar", text),
        new Promise((res) => setTimeout(res, 1500))
      ]);
    }
  } catch { /* نكمل بالخط الاحتياطي */ }
}

async function retarget() {
  await loadFont();
  sampleTargets();
}

/* ---------- التفاعل ---------- */
function setMouse(e) {
  const r = canvas.getBoundingClientRect();
  mouse.x = (e.clientX - r.left) * dpr;
  mouse.y = (e.clientY - r.top) * dpr;
}
if (!reduceMotion) {
  canvas.addEventListener("pointermove", setMouse);
  canvas.addEventListener("pointerdown", (e) => { setMouse(e); mouse.power = 32 * dpr; });
  canvas.addEventListener("pointerleave", () => { mouse.x = mouse.y = -9999; });
}

let typing = 0;
$("msg").addEventListener("input", (e) => {
  clearTimeout(typing);
  typing = setTimeout(() => {
    text = e.target.value.trim() || DEFAULT_TEXT;
    retarget();
  }, 160);
});

let resizing = 0;
window.addEventListener("resize", () => {
  clearTimeout(resizing);
  resizing = setTimeout(() => { fitCanvas(); sampleTargets(); }, 200);
});

/* ---------- النصائح ---------- */
let lastTip = -1;
function nextTip() {
  let i;
  do { i = Math.floor(Math.random() * TIPS.length); } while (i === lastTip && TIPS.length > 1);
  lastTip = i;
  $("tip").textContent = TIPS[i];
  const box = $("tipBox");
  box.classList.remove("in");
  void box.offsetWidth;            // لإعادة تشغيل الحركة
  box.classList.add("in");
}
$("more").addEventListener("click", nextTip);

/* ---------- البداية ---------- */
if (MY_NAME) {
  const c = $("credit");
  c.textContent = "برمجة: " + MY_NAME;
  c.hidden = false;
}
nextTip();
fitCanvas();
retarget().then(() => {
  if (!reduceMotion && !running) { running = true; tick(); }
});
</script>
